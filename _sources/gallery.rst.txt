   function twFilter(level) {
     document.querySelectorAll('.tw-filter').forEach(b => b.classList.remove('active'));
     event.target.classList.add('active');
     document.querySelectorAll('.tw-gallery-card').forEach(card => {
       card.style.display = (level === 'all' || card.dataset.level === level) ? '' : 'none';
     });
   }
   </script>
